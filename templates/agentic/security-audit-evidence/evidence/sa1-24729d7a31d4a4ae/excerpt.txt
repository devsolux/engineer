SLUG_KEEP = re.compile(r"[^a-z0-9-]+")

def slugify(title):
    slug = SLUG_KEEP.sub("-", title.lower()).strip("-")
    return slug or "note"

def export_path(export_dir, title):
    return os.path.join(export_dir, slugify(title) + ".md")

def write_export(export_dir, note):
    with open(export_path(export_dir, note.title), "w", encoding="utf-8") as handle:
        handle.write(note.body)
